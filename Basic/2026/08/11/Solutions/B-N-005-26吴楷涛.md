#include <stdio.h>

int main(void)
{
    int t;
    scanf("%d",&t);
    for(int i=0;i<t;i++)
    {
        int n;
        scanf("%d",&n);
        int w[1005]={0};
        int total=0;
        for(int j=0;j<n;j++)
        {
            int l;
            scanf("%d",&l);
            w[l]++;
            total += l; 
        }
        int biao = 0;
        int sum = 0;
        int max = 0;    
        for(int j=1;j<=1000;j++)
        {
            if(w[j]>0)
            {
                sum += j;
                if(w[j]>=2)
                {
                    biao = 1;
                    if(j>max)
                    {
                        max = j;
                    }
                }
            }
        }
        int x;
        if(!biao)
        {
            x = total;
        }
        else
        {
            x = sum + max*2;
        }
        printf("%d\n",x);
    }
    return 0;
}
