#include <stdio.h>
#include <limits.h>

int main(void)
{
    int t,n;
    long long s[110];
    
    scanf("%d",&t);
    for(int i=0;i<t;i++)
    {
        scanf("%d",&n);
        for(int j=0;j<n;j++)
        {
            scanf("%lld",&s[j]);
        }
        if(n%2!=0)
        {
            printf("NO\n");
            continue;
        }
        long long max = LLONG_MAX;
        long long min = LLONG_MIN;
        for(int d=0;d<n;d++)
        {
            int u=d+1;
            if(u%2==0)
            {
                if(s[d]>min)
                {
                    min = s[d];
                }
            }
            else
            {
                if(s[d]<max)
                {
                    max = s[d];
                }
            }
        }
        if(max-min<=2)
        {
            printf("YES\n");
        }
        else 
        {
            printf("NO\n");
        }
    }
    return 0;
}
