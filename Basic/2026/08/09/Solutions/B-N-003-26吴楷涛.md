#include <stdio.h>
int main(void)
{
    int T,n,k;
    int vi[1005];
    scanf("%d",&T);
    for(int i=0;i<T;i++)
    {
        scanf("%d %d",&n,&k);
        for(int o=0;o<n;o++)
        {
            scanf("%d ",&vi[o]);
        }
        for(int y=0;y<n-1;y++)
        {
            for(int j=0;j<n-1-y;j++)
            {
                if(vi[j]<vi[j+1])
                {
                    int t = vi[j];
                    vi[j] = vi[j+1];
                    vi[j+1] = t;
                }
            }
        }
        int mid_s[1005];
        for(int s=0;s<k;s++)
        {
            mid_s[s] = vi[s];
        }
        for(int u=0;u<k-1;u++)
        {
            for(int b=0;b<k-1-u;b++)
            {
                if(mid_s[b]>mid_s[b+1])
                {
                    int h = mid_s[b];
                    mid_s[b] = mid_s[b+1];
                    mid_s[b+1] = h;
                }
            }
        }
        int Max = mid_s[k-1];
        int m = (k+1)/2;
        int Mid = mid_s[m-1];
        printf("%d\n",Max-Mid);
    }
    return 0;
}
