#include <stdio.h>
int main(void)
{
    int N;
    int s[105];
    scanf("%d",&N);
    for(int i=0;i<N;i++)
    {
        scanf("%d ",&s[i]);
    }
    for(int j=0;j<N-1;j++)
    {
        for(int k=0;k<N-1-j;k++)
        {
            if(s[k]>s[k+1])
            {
                int y = s[k+1];
                s[k+1] = s[k];
                s[k] = y;
            }
        }
    }
    int t=0;
    for(int f=0;f<N;f++)
    {
        if(t==0 || s[f]!=s[t-1])
        {
            s[t++] = s[f];
        }
    }
    printf("%d\n",t);
    for(int h=0;h<t;h++)
    {
        for(int g=0;g<t-1-h;g++)
        {
            if(s[g]>s[g+1])
            {
                s[g+1] = s[g];
            }
        }
    }
    for(int l=0;l<t;l++)
    {
        if(l>0)
        {
            printf(" ");
        }
        printf("%d ",s[l]);
    }
    return 0;
}
