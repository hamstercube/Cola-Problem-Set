#include <stdio.h>
#include <string.h>
#define MIT 200010
char t[MIT];
int k = 0;
int main()
{
    char s[MIT];
    int n;
    scanf("%d",&n);
    scanf("%s",s);
    k=0;
    for(int i=0;i<n;i++)
    {
        t[k++]=s[i];
        if(k>=3 && t[k-3]=='n' && t[k-2]=='i' && t[k-1]=='u')
        {
            k-=3;
        }
        else if(k>=2 && t[k-2]=='m' && t[k-1]=='o')
        {
            k-=2;
        }
    }
    if(k==0)
    {
        printf("Yes");
    }
    else
    {
        printf("No");
    }
    return 0;
}
