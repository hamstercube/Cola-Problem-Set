#include <stdio.h>
int main(void)
{
    int n;
    int s[100];
    scanf("%d",&n);
    for(int i=0;i<n;i++)
    {
        scanf("%d",&s[i]);
    }
    int min = s[0];
    int max = s[0];
    for(int j=0;j<n;j++)
    {
        if(min>s[j])
        {
            min = s[j];
        }
        if(max<s[j])
        {
            max = s[j];
        }
    }
    if(min==max)
    {
        printf("1 0\n");
        return 0;
    }
    int mi=0,ma=0;
    int m=0;
    for(int k = 0;k<n;k++)
    {
        if(s[k]==min)
        {
            mi++;
        }
        else if(s[k]==max)
        {
            ma++;
        }
        else
        {
            m=1;
        }
    }
    int z= max-min;

}
